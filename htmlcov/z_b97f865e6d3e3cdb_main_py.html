72@app.exception_handler(HTTPException) 

73async def http_exception_handler( 

74 request: Request, 

75 exc: HTTPException, 

76): 

77 """Перетворює HTTPException у єдиний формат ErrorResponse.""" 

78 

79 if isinstance(exc.detail, dict) and "code" in exc.detail: 

80 content = { 

81 "code": exc.detail.get("code"), 

82 "message": exc.detail.get("message"), 

83 "details": exc.detail.get("details"), 

84 } 

85 else: 

86 content = { 

87 "code": "HTTP_ERROR", 

88 "message": str(exc.detail), 

89 "details": None, 

90 } 

91 

92 return JSONResponse( 

93 status_code=exc.status_code, 

94 content=content, 

95 headers=exc.headers, 

96 )

99@app.exception_handler(RequestValidationError) 

100async def validation_exception_handler( 

101 request: Request, 

102 exc: RequestValidationError, 

103): 

104 """Перетворює помилки валідації FastAPI/Pydantic у ErrorResponse.""" 

105 

106 details = [] 

107 

108 for error in exc.errors(): 

109 details.append( 

110 { 

111 "field": ".".join(str(part) for part in error["loc"]), 

112 "message": error["msg"], 

113 "type": error["type"], 

114 } 

115 ) 

116 

117 return JSONResponse( 

118 status_code=422, 

119 content={ 

120 "code": "VALIDATION_ERROR", 

121 "message": "Помилка валідації вхідних даних", 

122 "details": details, 

123 }, 

124 )
